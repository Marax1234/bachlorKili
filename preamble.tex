  \ifinteger{#1}
    {#1\adddot\space Auflage}
    {#1}}

\renewcommand{\newunitpunct}{\addcomma\space}

% Strings
\DefineBibliographyStrings{ngerman}{%
  in       = {In},
  urlseen  = {zuletzt geprüft am},
  editor   = {Hrsg\adddot},
  editors  = {Hrsg\adddot},
}

% "In:" mit Doppelpunkt; Punkt vor "In:" statt Komma
\renewcommand{\intitlepunct}{\addcolon\space}
\renewbibmacro*{in:}{%
  \setunit{\adddot\space}%
  \printtext{In}%
  \intitlepunct
}

% Seitenangaben ohne "S."
\DeclareFieldFormat{pages}{#1}

% Volume/Number für Zeitschriften: "Zeitschrift, 50 (4), Seiten"
\DeclareFieldFormat[article]{volume}{#1}
\DeclareFieldFormat[article]{number}{\mkbibparens{#1}}
\renewbibmacro*{volume+number+eid}{%
  \printfield{volume}%
  \setunit*{\addnbspace}%
  \printfield{number}%
  \setunit{\addcomma\space}%
  \printfield{eid}%
}
% Komma zwischen Zeitschriftenname und Volume (statt Leerzeichen)
\renewbibmacro*{journal+issuetitle}{%
  \usebibmacro{journal}%
  \setunit*{\addcomma\space}%
  \usebibmacro{volume+number+eid}%
  \newunit
}

% Ort, Verlag (Komma statt Doppelpunkt), kein Datum (steht im Label)
\renewbibmacro*{publisher+location+date}{%
  \printlist{location}%
  \iflistundef{publisher}
    {}
    {\setunit*{\addcomma\space}\printlist{publisher}}%
  \newunit
}

% Datum: nur Jahr im Label, Tag/Monat unterdrücken
\AtEveryBibitem{%
  \clearfield{day}%
  \clearfield{month}%
  \clearfield{endday}%
  \clearfield{endmonth}%
}

% URLs in Normalschrift (sans Mono) setzen
\urlstyle{same}

% URL + Zugriffsdatum: "URL: http://... (zuletzt geprüft am DD.MM.YYYY)"
\DeclareFieldFormat{url}{URL\addcolon\space\url{#1}}
\renewbibmacro*{url+urldate}{%
  \printfield{url}%
  \iffieldundef{urlyear}
    {}
    {\setunit*{\addspace}%
     \printtext{\mkbibparens{zuletzt geprüft am\addspace
       \iffieldundef{urlday}
         {\thefield{urlyear}}
         {\ifnumless{\thefield{urlday}}{10}{0}{}\thefield{urlday}.%
          \ifnumless{\thefield{urlmonth}}{10}{0}{}\thefield{urlmonth}.%
          \thefield{urlyear}}%
     }}}%
  \newunit
}

% Editor-String in Klammern: "Name, Vorname (Hrsg.)" statt "Name, Vorname, Hrsg."
\DeclareDelimFormat{editortypedelim}{\addspace}

% byeditor+others: leer (Editor wird im maintitle+booktitle-Makro behandelt)
\renewbibmacro*{byeditor+others}{}

% KSH incollection: "In: Editor (Hrsg.): Buchtitel" statt "In: Buchtitel. Editor"
\renewbibmacro*{maintitle+booktitle}{%
  \ifnameundef{editor}
    {}
    {\printnames{editor}%
     \setunit*{\addspace}%
     \printtext{\mkbibparens{\bibstring{editor}}}%
     \setunit{\addcolon\space}}%
  \usebibmacro{booktitle}%
  \newunit
}
